-- 1. Criação da base de dados e tabelas
CREATE DATABASE loja_computadores;
USE loja_computadores;

CREATE TABLE Fabricantes (
    Codigo INT PRIMARY KEY,
    Nome VARCHAR(50)
);

CREATE TABLE Produtos (
    Codigo INT PRIMARY KEY,
    Nome VARCHAR(50),
    Preco DECIMAL(10, 2),
    Fabricante INT,
    FOREIGN KEY (Fabricante) REFERENCES Fabricantes(Codigo)
);

-- 6. Selecione o nome e o preço do produto mais barato.
SELECT Nome, Preco FROM Produtos ORDER BY Preco ASC LIMIT 1;

-- 7. Selecione o nome dos produtos com preço menor ou igual a 200.
SELECT Nome FROM Produtos WHERE Preco <= 200;

-- 8. Selecione todos os produtos com preço entre 60 e 120.
SELECT * FROM Produtos WHERE Preco BETWEEN 60 AND 120;

-- 9. Seleciona todos os produtos que contêm a nome “drive”;

-- 10. Selecione todos os produtos que terminam com a letra “r”
